cmake_minimum_required(VERSION 3.20)
project(online_index_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <malloc.h>
#ifndef __GLIBC__
#error glibc required
#endif
int main() { return malloc_trim(0); }" ENGINE_HAS_MALLOC_TRIM)
include(FetchContent)
option(ENGINE_BUNDLED_DEPS "Build pinned public dependencies" ON)
if(ENGINE_BUNDLED_DEPS)
 set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
 set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
 FetchContent_Declare(abseil URL https://codeload.github.com/abseil/abseil-cpp/tar.gz/refs/tags/20240116.2 URL_HASH SHA256=733726b8c3a6d39a4120d7e45ea8b41a434cdacde401cba500f14236c49b39dc)
 FetchContent_MakeAvailable(abseil)
 set_property(DIRECTORY "${abseil_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
 set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(protobuf URL https://codeload.github.com/protocolbuffers/protobuf/tar.gz/refs/tags/v21.12)
 FetchContent_MakeAvailable(protobuf)
 set_property(DIRECTORY "${protobuf_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
else()
 find_package(Protobuf CONFIG REQUIRED)
 find_package(absl CONFIG REQUIRED)
endif()
set(FLATBUFFERS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATC OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(flatbuffers URL https://codeload.github.com/google/flatbuffers/tar.gz/refs/tags/v2.0.8 URL_HASH SHA256=f97965a727d26386afaefff950badef2db3ab6af9afe23ed6d94bfb65f95f37e)
FetchContent_MakeAvailable(flatbuffers)
set_property(DIRECTORY "${flatbuffers_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/proto)
set(PROTO_FILES ${CMAKE_CURRENT_SOURCE_DIR}/core/proto/header.proto ${CMAKE_CURRENT_SOURCE_DIR}/core/proto/snapshot_meta.proto)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/proto/header.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/proto/header.pb.h ${CMAKE_CURRENT_BINARY_DIR}/proto/snapshot_meta.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/proto/snapshot_meta.pb.h
 COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR} --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/core ${PROTO_FILES}
 DEPENDS protobuf::protoc ${PROTO_FILES} VERBATIM)
add_library(engine_proto ${CMAKE_CURRENT_BINARY_DIR}/proto/header.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/proto/snapshot_meta.pb.cc)
target_link_libraries(engine_proto PUBLIC protobuf::libprotobuf)
target_include_directories(engine_proto PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
add_library(engine_core src/flags.cc)
target_compile_definitions(engine_core PUBLIC ENGINE_HAVE_MALLOC_TRIM=$<BOOL:${ENGINE_HAS_MALLOC_TRIM}>)
target_include_directories(engine_core PUBLIC compat core src ${flatbuffers_SOURCE_DIR}/include)
target_link_libraries(engine_core PUBLIC engine_proto absl::flat_hash_map absl::hash Threads::Threads)
option(ENGINE_SANITIZE "Address and undefined behavior checks" OFF)
if(ENGINE_SANITIZE)
 target_compile_options(engine_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(engine_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(online_index src/main.cc)
target_link_libraries(online_index PRIVATE engine_core)
add_executable(online_index_worker src/worker.cc)
target_link_libraries(online_index_worker PRIVATE engine_core)

# When built inside the combined project, use its existing batch-query target.
# A standalone build can point to a separate KVQueryEngine checkout instead.
set(ENGINE_KVQUERY_SOURCE_DIR "" CACHE PATH "Path to a KVQueryEngine checkout")
if(TARGET kvquery)
 target_compile_definitions(online_index_worker PRIVATE ENGINE_HAVE_KVQUERY=1)
 target_link_libraries(online_index_worker PRIVATE kvquery)
elseif(ENGINE_KVQUERY_SOURCE_DIR)
 if(NOT EXISTS "${ENGINE_KVQUERY_SOURCE_DIR}/src/kvquery.cpp" OR
    NOT EXISTS "${ENGINE_KVQUERY_SOURCE_DIR}/include/kvquery/kvquery.hpp")
  message(FATAL_ERROR "ENGINE_KVQUERY_SOURCE_DIR is not a KVQueryEngine checkout")
 endif()
 add_library(engine_kvquery STATIC "${ENGINE_KVQUERY_SOURCE_DIR}/src/kvquery.cpp")
 target_include_directories(engine_kvquery PUBLIC "${ENGINE_KVQUERY_SOURCE_DIR}/include")
 target_compile_features(engine_kvquery PUBLIC cxx_std_20)
 if(ENGINE_SANITIZE)
  target_compile_options(engine_kvquery PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 endif()
 target_compile_definitions(online_index_worker PRIVATE ENGINE_HAVE_KVQUERY=1)
 target_link_libraries(online_index_worker PRIVATE engine_kvquery)
endif()
