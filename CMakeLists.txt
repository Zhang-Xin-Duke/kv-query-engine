cmake_minimum_required(VERSION 3.20)
project(KVQueryEngine VERSION 0.1.0 LANGUAGES CXX)
option(KVQUERY_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(KVQUERY_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(KVQUERY_SANITIZERS AND KVQUERY_TSAN)
  message(FATAL_ERROR "ASan/UBSan and TSan require separate builds")
endif()
add_library(kvquery src/kvquery.cpp)
target_include_directories(kvquery PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(kvquery PUBLIC cxx_std_20)
target_compile_options(kvquery PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(KVQUERY_SANITIZERS)
  target_compile_options(kvquery PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(kvquery PUBLIC -fsanitize=address,undefined)
elseif(KVQUERY_TSAN)
  target_compile_options(kvquery PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(kvquery PUBLIC -fsanitize=thread)
endif()
find_package(Threads REQUIRED)
include(CTest)
if(BUILD_TESTING)
  add_executable(kvquery_test tests/kvquery_test.cpp)
  target_link_libraries(kvquery_test PRIVATE kvquery Threads::Threads)
  foreach(group edges collisions randomized concurrent persistence retry format)
    add_test(NAME kvquery_${group} COMMAND kvquery_test ${group})
  endforeach()
endif()
add_executable(kvquery_cli tools/kvquery_cli.cpp)
set_target_properties(kvquery_cli PROPERTIES OUTPUT_NAME kvquery)
target_link_libraries(kvquery_cli PRIVATE kvquery)
add_executable(kvquery_bench bench/kvquery_bench.cpp)
target_link_libraries(kvquery_bench PRIVATE kvquery)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME kvquery_cli_roundtrip COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:kvquery_cli>)
  add_test(NAME kvquery_benchmark_smoke COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_test.py $<TARGET_FILE:kvquery_bench>)
endif()
